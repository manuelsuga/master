-- Creación del KEYSPACE

CREATE KEYSPACE IF NOT EXISTS logistica_flotas 
WITH REPLICATION = {
	'class' : 'NetworkTopologyStrategy', 
	'datacenter1' : 3 
};

-- Creación de las tablas

CREATE TABLE logistica_flotas.info_rutas_por_conductor (
	idConductor text,
	idRuta int,
	fechaRuta date,
	almacenOrigen text,
	almacenDestino text,
	horasEstimadas int,
	PRIMARY KEY (idConductor, idRuta)
);

CREATE TABLE logistica_flotas.incidencias_conductores_por_ruta (
	idRuta int,
	idConductor text,
	incidencias map<text, text>,
	PRIMARY KEY (idRuta, idConductor)
) WITH CLUSTERING ORDER BY (idConductor ASC);
