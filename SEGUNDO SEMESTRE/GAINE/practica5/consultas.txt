// Consulta 1
MATCH (c:Club) ORDER BY c.nombre ASC RETURN c.nombre AS Club, c.ciudad AS Ciudad;

// Consulta 2
MATCH (c:Club)-[:JUEGA_EN_ESTADIO]->(e:Estadio) RETURN c.nombre, e.nombre;

// Consulta 3
MATCH (e:Entrenador)-[:ENTRENA_A]->(c:Club) ORDER BY c.nombre DESC RETURN e.nombre, c.nombre;

// Consulta 4
MATCH (j:Jugador)-[:JUEGA_EN]->(c:Club) RETURN COUNT(j) as total, c.nombre ORDER BY total ASC;

// Consulta 5
MATCH (p:Partido)-[:EQUIPO_LOCAL|EQUIPO_VISITANTE]->(c:Club) WHERE c.nombre='Albacete Balompie' RETURN p.partidoId;

// Consulta 6
MATCH (j:Jugador)-[:JUEGA_EN|JUGO_EN]->(c:Club) WITH j AS jugador, collect(DISTINCT c.nombre) AS clubes WHERE size(clubes) > 1 RETURN jugador.nombre, clubes ORDER BY size(clubes) DESC;

// Consulta 7
MATCH (j:Jugador)-[r:JUGO_PARTIDO]->(p:Partido) RETURN j.nombre, sum(r.goles) as totalGoles ORDER BY totalGoles DESC LIMIT 1;

// Consulta 8
MATCH (j:Jugador)-[r1:JUEGA_EN]->(c:Club) WITH j as jugador, c as clubActual MATCH (jugador)-[:JUGO_EN]->(clubActual) MATCH (jugador)-[:JUGO_EN|JUEGA_EN]->(c:Club) WITH jugador, collect(c.nombre) as historicoClubes  RETURN jugador.nombre, historicoClubes;

// Consulta 9
MATCH (j:Jugador)-[r:JUGO_PARTIDO]->(p:Partido) MATCH (j)-[:JUGO_EN]->(exClub:Club) MATCH (p)-[:EQUIPO_LOCAL|EQUIPO_VISITANTE]->(exClub) MATCH (j)-[:JUEGA_EN]->(clubActual:Club) WHERE r.goles > 0 AND exClub <> clubActual RETURN DISTINCT j.nombre, exClub.nombre;

MATCH (c:Cuenta)-[:REGISTRADA]->(p:Pais {paraiso_fiscal: true}) MATCH (pr:Persona)-[:TITULAR]->(c) WHERE c.balance > 10000 RETURN c.iban, c.balance, p.nombre, pr.nombre ORDER BY c.balance DESC;

MATCH (origen:Cuenta)-[t:TRANSFIERE]->(destino:Cuenta)-[:REGISTRADA]->(p:Pais {paraiso_fiscal: true}) RETURN origen.iban, destino.iban, t.importe, p.nombre;

MATCH (p1:Persona)-[:USA]->(md:Medio) MATCH (p2:Persona)-[:USA]->(md) WHERE p1 <> p2 WITH COLLECT(DISTINCT p2.nombre) as personas, md, p1 RETURN p1.nombre, personas, md.valor AS medio_compartido;

MATCH (pe:Persona {dni: "12345678A"}) OPTIONAL MATCH (pe)-[:TITULAR]->(c:Cuenta)-[:REGISTRADA]->(pais:Pais) OPTIONAL MATCH (pe)-[:USA]->(md:Medio) RETURN pe.nombre, pe.dni, collect(DISTINCT c.iban) AS cuentas, collect(DISTINCT pais.nombre) AS paises, collect(DISTINCT md.valor) AS medios;

MATCH (origen:Cuenta)-[t:TRANSFIERE]->(destino:Cuenta) WITH origen, count(t) AS num_transferencias, sum(t.importe) AS total, collect(t.importe) AS importes WHERE num_transferencias >= 2 RETURN origen.iban, num_transferencias, total, importes ORDER BY total DESC;
